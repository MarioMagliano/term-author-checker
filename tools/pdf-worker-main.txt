// ============================================================================
// WORKER-KLEBER: PDF-Verarbeitung im Browser (Web Worker)
// ----------------------------------------------------------------------------
// Diese Datei hängt "tools/build-pdf-worker.cjs" hinter die PDF-Module
// (PdfInflate, PdfTextPosition, PdfAnnotate, PdfShrink, PdfAnnotateSearch). Sie
// stellt nur her, was die Module aus Apps Script gewohnt sind:
//  - einen Leser, der Bytebereiche der PDF in Drive per HTTP-Range holt
//    (synchron, das geht im Worker; die Module sind synchron geschrieben),
//  - Logger und _ct_ (Beschriftung der Notizen),
//  - eine kleine Nachrichten-Schnittstelle zur Seite.
// Die Datei hat bewusst die Endung .txt: clasp soll sie nicht als Apps-Script-
// Code hochladen (sie greift auf "self" zu).
// ============================================================================

var Logger = { log: function() {} };
var _ctLabels_ = {};
function _ct_(key) { return Object.prototype.hasOwnProperty.call(_ctLabels_, key) ? _ctLabels_[key] : key; }

var BROWSER_BLOCK = 8 * 1024 * 1024;   // Größe einer Range-Anfrage (wie PDF_SHRINK_CHUNK)
var BROWSER_CACHE_BLOCKS = 3;          // so viele Blöcke bleiben im Speicher
var BROWSER_RETRIES = 4;

function _bytesToBinaryString_(u8) {
  var CH = 8192, parts = [];
  for (var i = 0; i < u8.length; i += CH) parts.push(String.fromCharCode.apply(null, u8.subarray(i, Math.min(i + CH, u8.length))));
  return parts.join('');
}

function _binaryStringToU8_(s) {
  var out = new Uint8Array(s.length);
  for (var i = 0; i < s.length; i++) out[i] = s.charCodeAt(i) & 255;
  return out;
}

// FNV-1a über die Zeichen (zum Vergleichen großer Texte in Tests).
function _fnv_(s) {
  var h = 0x811c9dc5;
  for (var i = 0; i < s.length; i++) { h ^= s.charCodeAt(i) & 255; h = Math.imul(h, 0x01000193) >>> 0; }
  return h >>> 0;
}

function _busyWait_(ms) { var end = Date.now() + ms; while (Date.now() < end) { /* kurze Pause zwischen Wiederholungen */ } }

/**
 * Leser (read/readMany/size wie _pdfDriveRangeReader_), gespeist per
 * synchroner XMLHttpRequest. cfg: { url, token, size, block? }.
 * Gelesen wird in festen Blöcken; ein kleiner Cache sorgt dafür, dass
 * aufeinanderfolgende kleine Lesungen (die Objekte der PDF) nur eine Anfrage
 * kosten. Bilder werden dadurch zwar mitgeladen, aber nie behalten - die Datei
 * wird einmal durchgelesen, nicht komplett im Speicher gehalten.
 */
function makeRangeReader_(cfg) {
  var block = cfg.block || BROWSER_BLOCK;
  var cache = {}, order = [];
  var stats = { requests: 0, bytes: 0, retries: 0 };

  // Blöcke bleiben als Bytes (Uint8Array) im Cache und werden erst beim Lesen in
  // einen Text umgewandelt. Ein Teilstring (slice) eines großen Textes hielte den
  // ganzen Block im Speicher - bei einer Datei mit vielen Bildern also die ganze
  // Datei.
  function fetchBlock(idx) {
    var start = idx * block, end = Math.min(cfg.size, start + block) - 1;
    for (var attempt = 0; ; attempt++) {
      var xhr = new XMLHttpRequest();
      try {
        xhr.open('GET', cfg.url, false);
        xhr.responseType = 'arraybuffer';
        if (cfg.token) xhr.setRequestHeader('Authorization', 'Bearer ' + cfg.token);
        xhr.setRequestHeader('Range', 'bytes=' + start + '-' + end);
        xhr.send(null);
        stats.requests++;
        if (xhr.status === 401) { var e401 = new Error('Drive session expired (HTTP 401). Please start again.'); e401.fatal = true; throw e401; }
        if (xhr.status === 206 || xhr.status === 200) {
          var u8 = new Uint8Array(xhr.response);
          // Range ignoriert (HTTP 200 mit der ganzen Datei): passenden Teil nehmen.
          if (xhr.status === 200 && u8.length > end - start + 1) u8 = u8.slice(start, end + 1);
          stats.bytes += u8.length;
          return u8;
        }
        if (xhr.status === 403 || xhr.status === 404) { var eF = new Error('Drive download failed (HTTP ' + xhr.status + ').'); eF.fatal = true; throw eF; }
        throw new Error('Drive download failed (HTTP ' + xhr.status + ').');
      } catch (err) {
        if (err.fatal || attempt >= BROWSER_RETRIES) throw err;
        stats.retries++;
        _busyWait_(400 * Math.pow(2, attempt));
      }
    }
  }

  function getBlock(idx) {
    if (Object.prototype.hasOwnProperty.call(cache, idx)) {
      order.splice(order.indexOf(idx), 1); order.push(idx);
      return cache[idx];
    }
    var u8 = fetchBlock(idx);
    cache[idx] = u8; order.push(idx);
    while (order.length > BROWSER_CACHE_BLOCKS) delete cache[order.shift()];
    return u8;
  }

  function read(start, end) {
    end = Math.min(end, cfg.size);
    if (start >= end) return '';
    var first = Math.floor(start / block), last = Math.floor((end - 1) / block);
    if (first === last) {
      return _bytesToBinaryString_(getBlock(first).subarray(start - first * block, end - first * block));
    }
    var out = [];
    for (var i = first; i <= last; i++) {
      var from = Math.max(start, i * block) - i * block, to = Math.min(end, (i + 1) * block) - i * block;
      out.push(_bytesToBinaryString_(getBlock(i).subarray(from, to)));
    }
    return out.join('');
  }

  return {
    size: cfg.size,
    stats: stats,
    read: read,
    readMany: function(ranges) { return ranges.map(function(r) { return read(r[0], r[1]); }); }
  };
}

// ─── Zustand und Befehle ──────────────────────────────────────────────────
var S = { reader: null, an: null, cfg: null };

function post_(msg, transfer) { self.postMessage(msg, transfer || []); }

function cmdOpen_(m) {
  var t0 = Date.now();
  S.cfg = { url: m.url, token: m.token, size: m.size, block: m.block };
  S.reader = makeRangeReader_(S.cfg);
  S.an = _driveAnnotatedAnalyze_(S.reader);
  var text = S.an.model.text;
  return {
    pages: S.an.pages.length, encrypted: !!S.an.model.encrypted,
    imagesRemoved: S.an.model.imagesRemoved, imagesKept: S.an.model.imagesKept,
    shrunkBytes: text.length, shrunkHash: _fnv_(text),
    requests: S.reader.stats.requests, downloadedBytes: S.reader.stats.bytes, retries: S.reader.stats.retries,
    ms: Date.now() - t0
  };
}

function cmdAnnotate_(m) {
  if (!S.an) throw new Error('No PDF opened.');
  var t0 = Date.now();
  _ctLabels_ = m.labels || {};
  _driveAnnotatedRequirePlain_(S.an);
  var state = null, res = null, slices = 0;
  do {
    res = _drivePdfComputeAnnotations_(S.an.doc, S.an.pages, m.issues, null,
      { resume: state, stopAt: Date.now() + 300 });
    slices++;
    if (!res.done) {
      state = res.state;
      post_({ id: m.id, progress: { phase: 'search', next: state.next, count: res.count } });
    }
  } while (!res.done);
  var appendix = _driveAnnotatedAppendix_(S.reader, S.an, res.pageAnnotations);
  var u8 = _binaryStringToU8_(appendix);
  return {
    result: { count: res.count, total: res.total, positioned: res.positioned, slices: slices,
              appendixBytes: u8.length, ms: Date.now() - t0,
              requests: S.reader.stats.requests, downloadedBytes: S.reader.stats.bytes },
    transfer: [u8.buffer], appendix: u8
  };
}

// Bereitet die Prüfpakete vor (wie _drivePdfPrepUnit_ in Apps Script): Bilder bis
// zum Budget behalten, Seiten in Pakete teilen. Pakete bleiben im Worker; über
// "part" wird eines abgeholt.
function cmdSplit_(m) {
  var t0 = Date.now();
  var reader = S.reader || makeRangeReader_(m);
  var model = pdfRebuild_(reader, { imageBudget: m.imageBudget === undefined ? 12 * 1024 * 1024 : m.imageBudget });
  var splitter = pdfPageSplitter_({ text: model.text, offsets: model.offsets, compressed: model.compressed,
                                    rootNum: model.rootNum, maxNum: model.maxNum, encrypted: model.encrypted });
  var n = splitter.pageCount, per = m.pagesPerPart || 4, sizes = [];
  S.parts = [];
  for (var from = 0; from < n; from += per) {
    var piece = splitter.build(from, Math.min(n, from + per));
    S.parts.push(piece);
    sizes.push(piece.length);
  }
  return { result: { pages: n, parts: S.parts.length, sizes: sizes, shrunkBytes: model.text.length,
                     imagesKept: model.imagesKept, imagesRemoved: model.imagesRemoved,
                     requests: reader.stats.requests, downloadedBytes: reader.stats.bytes, ms: Date.now() - t0 } };
}

function cmdPart_(m) {
  var u8 = _binaryStringToU8_(S.parts[m.index]);
  return { result: { bytes: u8.length }, transfer: [u8.buffer], part: u8 };
}

self.onmessage = function(ev) {
  var m = ev.data, out;
  try {
    if (m.cmd === 'open') out = { result: cmdOpen_(m) };
    else if (m.cmd === 'annotate') out = cmdAnnotate_(m);
    else if (m.cmd === 'split') out = cmdSplit_(m);
    else if (m.cmd === 'part') out = cmdPart_(m);
    else throw new Error('Unknown command: ' + m.cmd);
  } catch (err) {
    post_({ id: m.id, error: { message: String(err && err.message || err), fatal: !!(err && err.fatal), structure: !!(err && err.pdfStructure) } });
    return;
  }
  var reply = { id: m.id, result: out.result };
  if (out.appendix) reply.appendix = out.appendix;
  if (out.part) reply.part = out.part;
  post_(reply, out.transfer);
};
